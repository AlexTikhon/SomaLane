// Post-hoc deterministic probe (supplied claims, no model): does the production grounding accept a CORRECT claim for
// the values and titles the real model missed? Not a gate input; changes nothing.
import { resolve } from 'node:path';
const api = process.cwd();
const imp = (p) => import(new URL(`file:///${resolve(api, p).replaceAll('\\', '/')}`).href);
const { groundMealDraft } = await imp('src/meals/drafts/meal-draft-grounding.ts');
const { mealDraftWireSchema } = await imp('src/meals/drafts/meal-draft-wire.ts');

const ns = { value: null, evidence: null, abstain: 'not_stated' };
const claim = (value, evidence) => ({ value, evidence, abstain: null });
const probe = (label, text, answer) => {
  const wire = mealDraftWireSchema.parse({ outcome: 'draft', mealType: null, mealTypeEvidence: null, energyKcal: ns, proteinG: ns, carbohydrateG: ns, fatG: ns, ...answer });
  try {
    const { proposal } = groundMealDraft(text, wire);
    const n = Object.fromEntries(Object.entries(proposal.nutrition ?? {}).map(([k, v]) => [k, v.status === 'proposed' ? v.value : v.status === 'withheld' ? `withheld:${v.reason}` : v.status]));
    console.log(`${label}: ${proposal.outcome} title=${JSON.stringify(proposal.title)} ${JSON.stringify(n)}`);
  } catch (error) {
    console.log(`${label}: ${error.constructor.name} ${error.message?.slice(0, 120) ?? ''}`);
  }
};

const soup = 'Tomato soup; the whole batch is 1200 kcal.\nIn total I ate 300 kcal and 9 g protein.';
probe('en-ho2-batch protein "9 g protein"', soup, { title: 'Tomato soup', energyKcal: claim('300', 'ate 300 kcal'), proteinG: claim('9', '9 g protein') });
probe('en-ho2-batch protein "300 kcal and 9 g protein"', soup, { title: 'Tomato soup', energyKcal: claim('300', '300 kcal'), proteinG: claim('9', '300 kcal and 9 g protein') });
probe('en-ho2-batch protein "protein 9 g" (not an excerpt)', soup, { title: 'Tomato soup', energyKcal: claim('300', '300 kcal'), proteinG: claim('9', 'protein 9 g') });

const cottage = 'For breakfast I had cottage cheese (150 g) with walnuts (30 g): 360 kcal and protein 25 g.';
for (const title of ['cottage cheese (150 g) with walnuts (30 g)', 'cottage cheese (150 g) with walnuts', 'cottage cheese', 'cottage cheese with walnuts']) {
  probe(`en-dev2-food-weights title ${JSON.stringify(title)}`, cottage, { title, mealType: 'breakfast', mealTypeEvidence: 'breakfast', energyKcal: claim('360', '360 kcal'), proteinG: claim('25', 'protein 25 g') });
}
const ryazhenka = 'Выпила ряженку: 134,4 ккал, жиры 5,25 г и белки 6,1 г';
for (const title of ['ряженку', 'Ряженка', 'ряженка']) {
  probe(`ru-dev2-decimal title ${JSON.stringify(title)}`, ryazhenka, { title, energyKcal: claim('134,4', '134,4 ккал'), proteinG: claim('6,1', 'белки 6,1 г') });
}
const beef = 'На ужин 200 г тушёной говядины и 150 г булгура. Всё вместе 560 ккал, белки 48 г.';
for (const title of ['200 г тушёной говядины и 150 г булгура', 'тушёной говядины и 150 г булгура', 'тушёной говядины', 'тушёная говядина и булгур', 'тушёной говядины и булгура']) {

}
